## Measurement script

```js
import { chromium } from '<a site checkout>/node_modules/playwright/index.mjs';
const urls = process.argv.slice(2);
const browser = await chromium.launch();
for (const url of urls) {
  const ctx = await browser.newContext({ viewport: { width: 412, height: 823 }, deviceScaleFactor: 1.75, isMobile: true, hasTouch: true });
  const page = await ctx.newPage();
  const bytes = new Map();
  page.on('response', async (r) => {
    if (r.request().resourceType() !== 'image') return;
    const len = Number(r.headers()['content-length'] || 0);
    let b = len; if (!b) { try { b = (await r.body()).length; } catch {} }
    bytes.set(r.url(), b);
  });
  await page.goto(url, { waitUntil: 'load' });
  // scroll to trigger lazy images
  await page.evaluate(async () => { for (let y = 0; y < document.body.scrollHeight; y += 400) { window.scrollTo(0, y); await new Promise(r => setTimeout(r, 120)); } window.scrollTo(0, 0); });
  await page.waitForTimeout(1500);
  const rows = await page.evaluate(async (imgUrls) => {
    const abs = (u) => new URL(u, location.href).href;
    const uses = new Map(); // url -> [{kind, w, h, size}]
    const add = (u, v) => { u = abs(u); if (!uses.has(u)) uses.set(u, []); uses.get(u).push(v); };
    for (const img of document.querySelectorAll('img')) {
      if (!img.currentSrc) continue;
      const r = img.getBoundingClientRect();
      add(img.currentSrc, { kind: 'img', w: r.width, h: r.height, fit: getComputedStyle(img).objectFit });
    }
    for (const el of document.querySelectorAll('*')) {
      for (const pseudo of [null, '::before', '::after']) {
        const cs = getComputedStyle(el, pseudo);
        const bg = cs.backgroundImage;
        if (!bg || bg === 'none') continue;
        for (const m of bg.matchAll(/url\("?([^")]+)"?\)/g)) {
          const r = el.getBoundingClientRect();
          add(m[1], { kind: 'bg' + (pseudo || ''), w: r.width, h: r.height, size: cs.backgroundSize });
        }
      }
    }
    const out = [];
    for (const u of imgUrls) {
      let nw = 0, nh = 0;
      try { const i = new Image(); i.src = u; await i.decode(); nw = i.naturalWidth; nh = i.naturalHeight; } catch {}
      const us = uses.get(u) || [];
      // width the image actually needs: for cover/object-fit cover, the scaled width
      let need = 0, kind = us.map(x => x.kind).join(',') || 'unused';
      for (const x of us) {
        let w = x.w;
        const cover = (x.size === 'cover') || (x.fit === 'cover');
        if (cover && nw && nh) w = Math.max(x.w, x.h * nw / nh);
        need = Math.max(need, w);
      }
      out.push({ u, nw, nh, kind, boxW: Math.max(0, ...us.map(x => x.w)), need });
    }
    return out;
  }, [...bytes.keys()]);
  console.log(`\n## ${url}`);
  for (const r of rows) {
    if (/\.svg(\?|$)/i.test(r.u)) continue;
    const need = r.need * 1.75;
    const ratio = need ? (r.nw / need) : NaN;
    console.log([new URL(r.u).pathname.slice(-60), r.kind, `${r.nw}x${r.nh}`, `box ${r.boxW.toFixed(0)}`, `need ${need.toFixed(0)}`, `ratio ${ratio.toFixed(2)}`, `${((bytes.get(r.u)||0)/1024).toFixed(0)}KB`].join(' | '));
  }
  await ctx.close();
}
await browser.close();
```
